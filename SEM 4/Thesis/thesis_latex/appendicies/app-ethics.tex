\noindent\textbf{Project Aim}

To develop a machine learning-based personalised voice bundle recommendation framework for Airtel Uganda's prepaid subscribers that improves customer satisfaction, increases bundle offer conversion rates, and informs data-driven revenue optimisation strategies.

\vspace{0.3cm}
\noindent\textbf{Objectives and their Expected Outputs}

\begin{enumerate}
    \item \textbf{Identify key customer behavioural attributes and patterns that are most predictive of voice bundle purchase decisions among Airtel Uganda prepaid subscribers.}
    \begin{itemize}
        \item Cleaned and anonymised transactional dataset
        \item Visual/statistical summaries of behavioural distributions
        \item Summary report of key behavioural attribute trends
    \end{itemize}

    \item \textbf{Test and evaluate suitable machine learning models --- including data preprocessing and feature engineering --- for accurately predicting voice bundle purchase decisions.}
    \begin{itemize}
        \item Preprocessed feature-engineered dataset
        \item Model comparison results (Logistic Regression, Decision Tree, Random Forest, XGBoost)
        \item Feature importance visualisations
    \end{itemize}

    \item \textbf{Evaluate and validate the selected recommendation model on a hold-out test set.}
    \begin{itemize}
        \item Final XGBoost model with Accuracy, Precision, Recall, F1-Score
        \item Evaluation report with benchmark comparisons
    \end{itemize}

    \item \textbf{Formulate strategic, evidence-based machine learning recommendations for Airtel Uganda to grow its business and improve customer satisfaction.}
    \begin{itemize}
        \item Strategic recommendation report with specific deployment actions
        \item Projected business impact analysis (conversion rate, ARPU uplift)
    \end{itemize}
\end{enumerate}
